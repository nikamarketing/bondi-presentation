# GitHub'da çalışan SEO audit script
# Claude bu scripti yazdı, biz çalıştırıyoruz

import requests
from bs4 import BeautifulSoup

def check_meta_tags(url):
    r = requests.get(url)
    soup = BeautifulSoup(r.text, 'html.parser')

    return {
        "title_length": len(soup.title.string),
        "has_meta_desc": bool(soup.find('meta', {'name':'description'})),
        "h1_count": len(soup.find_all('h1'))
    }
